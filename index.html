How to Enable Real-Time Auto-Saving to Google Sheets:

  1. Open your Google Sheet linked above.
  2. Click on Extensions ➔ Apps Script in the top menu.
  3. Delete any default code in the editor and paste the Google Apps Script code below.
  4. Click Deploy ➔ New Deployment.
  5. Select type: Web App. Set "Execute as": Me, and "Who has access": Anyone.
  6. Click Deploy and copy the generated Web App URL!
  7. Paste that Web App URL in the setup box below and click Save & Test Sync.
function doGet(e) { var sheet = SpreadsheetApp.getActiveSpreadsheet(); var data = { transactions: getSheetData(sheet, 'Transactions'), income: getSheetData(sheet, 'Income'), debts: getSheetData(sheet, 'Debts'), trips: getSheetData(sheet, 'Trips'), savingsGoals: getSheetData(sheet, 'SavingsGoals') }; return ContentService.createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); } function doPost(e) { var payload = JSON.parse(e.postData.contents); var sheet = SpreadsheetApp.getActiveSpreadsheet(); if (payload.action === 'sync') { writeSheetData(sheet, 'Transactions', payload.transactions); writeSheetData(sheet, 'Income', payload.income); writeSheetData(sheet, 'Debts', payload.debts); writeSheetData(sheet, 'Trips', payload.trips); writeSheetData(sheet, 'SavingsGoals', payload.savingsGoals); } return ContentService.createTextOutput(JSON.stringify({ status: 'success' })) .setMimeType(ContentService.MimeType.JSON); } function getSheetData(ss, name) { var s = ss.getSheetByName(name); if (!s) return []; var rows = s.getDataRange().getValues(); if (rows.length < 2) return []; var headers = rows[0]; var result = []; for (var i = 1; i < rows.length; i++) { var obj = {}; for (var j = 0; j < headers.length; j++) { obj[headers[j]] = rows[i][j]; } result.push(obj); } return result; } function writeSheetData(ss, name, dataList) { if (!dataList || dataList.length === 0) return; var s = ss.getSheetByName(name); if (!s) s = ss.insertSheet(name); s.clear(); var headers = Object.keys(dataList[0]); s.appendRow(headers); dataList.forEach(function(item) { var row = []; headers.forEach(function(h) { row.push(item[h] || ""); }); s.appendRow(row); }); }
